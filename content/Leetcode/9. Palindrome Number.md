class Solution {
    public boolean isPalindrome(int x) {

                /* i think there are two ways to do this
        first way is to reverse the given integer x so that everything is reversed. so base on definition of a palindrome
        which means even reverse and its the same it will be a palindrome, we compare the new reversed integer and see if its
        the same. but i dont know hash maps or anything so probably not this yet.

        the easier way is to use for loops and basically try to compare the first value with the last value and then keep repeating until you reach the center of the integer. so you only need to loop it x.length times because if it's an even number, it's perfect and if it's odd it's going to be same as the even number because the odd number means there will be a value in the middle and that middle value will always equal to itself so no need comparing.
        */


        /* this submission will be used to code for the first technique which is basically finding the String value of the integer x and then turning it into a character array using the toCharArray method for Strings in java. And then we can just loop for the length of the array and compare first index and last index and see if it's equal, and repeat for every index incrementing for the first number to compare and every index decrementing for the second number to compare until they reach the length of the array divide by 2.
        */

        boolean isPalindrome = true; // set the boolean to true for now.
        char[] char_array = String.valueOf(x).toCharArray();

        for (int i = 0; i < char_array.length / 2; i++) {
            if (char_array[i] != char_array[char_array.length-1-i]){
                isPalindrome = false;
                return isPalindrome;
            }
        }

        return isPalindrome;

        
    }
}
